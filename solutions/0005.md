# 0005 题解：语音克隆：NumPy 随机流复刻

> 配套题面：[problems/0005.md](../problems/0005.md)
> 验证代码：[verify/0005/solution.cpp](../verify/0005/solution.cpp)、[verify/0005/brute.cpp](../verify/0005/brute.cpp)

## 0. 出题来源

- 本地项目：`E:\project\musicclone\moss-tts-cpp`
- 远端仓库：<https://github.com/honghax/-MOSS-TTS-Nano-Cpp>（本人推送）
- 算法点：`cpp/src/main.cpp` 中为对齐 NumPy 而手写的 PCG64；原实现只固定了 `seed=1234` 的 state/inc。本题要求复刻 **SeedSequence(seed) → PCG64 → 跳转** 的完整链路。

这不是简单的「循环 K 次」：要先实现 NumPy 的 32 位种子混洗，再实现 128 位 LCG 的 $O(\log K)$ 跳转。

---

## 1. SeedSequence

完全按题面给出的常量与三轮混洗实现：

1. 把 seed 拆成 32 位小端字；
2. 用 `hashmix` 初始化 `pool[4]`；
3. 池内两两 `mix`；
4. 剩余熵字继续混入池；
5. 用 `INIT_B / MULT_B` 生成 8 个 32 位字，拼成 4 个 64 位字。

关键点：`hashmix` 的 `hash_const` 是**跨调用连续更新**的；`mix_entropy` 的第二轮是**原地**修改池，`src` 循环里读到的是已被前几轮修改过的 `pool[src]`。顺序错一位，结果就全错。

验证：`seed=1234` 得到的 `state/inc` 必须等于项目里写死的

```
state = 0x160ad84006fe21ea_f69b873d9fe45409
inc   = 0x50c8fb163c7cea4e_d0f51ce6006e4325
```

## 2. PCG64 初始化与跳转

初始化（模 $2^{128}$）：

```
inc = (initseq << 1) | 1
state = 0
state = state*MULT + inc
state += initstate
state = state*MULT + inc
```

取数：先 `state = state*MULT + inc`，再输出

```
out = rotr64(state.hi ^ state.lo, state.hi >> 58)
```

第 $K$ 次取数 = 从初始 state 步进 $K+1$ 次。步进是仿射映射 $A(x)=MULT\cdot x+INC$，对每个询问：

1. 用该 seed 的 `INC` 预处理 $A^{2^b}$（$b=0..63$）；
2. 把 $K+1$ 的二进制位对应的幂复合；
3. 作用到初始 state，输出。

复合公式：

$$
(B\circ A)(x)=a_B a_A x+a_B c_A+c_B .
$$

---

## 3. 正确性

- SeedSequence 部分逐行对应 NumPy 的 `mix_entropy` / `generate_state`，整数宽度与更新顺序一致；
- PCG 初始化与 `pcg_setseq_128_srandom_r` 一致；
- 仿射快速幂给出 $A^{K+1}$，作用到初始 state 后再取输出，正好是第 $K$ 次取数；
- 输出右移 11 位即项目 double 的分子。

---

## 4. 复杂度

- 每个询问：SeedSequence 约 20 次 32 位运算 + 64 次仿射平方 + 至多 64 次复合，即约 $O(\log K)$ 次 128 位乘法；
- $Q=10^5$：约 $10^7$ 次 128 位乘法，实测 **0.10 s**；
- 内存 $O(1)$。

---

## 5. 验证记录

- 样例 6 个询问全部匹配；
- 300 组小 $K$（≤5000）与 `brute.cpp`（SeedSequence + 顺序步进）**逐项一致**；
- 63 组随机大 $K$（$<10^{18}$）与 **NumPy 2.5.1** 的 `PCG64(seed).advance(K)` **逐项一致**；
- $Q=10^5$ 约 0.10 s。

## 6. 参考代码（C++17，已验证）

```cpp
// 0005 - NumPy default_rng(seed) 复刻：SeedSequence + PCG64 跳转
// 参考项目：E:\project\musicclone\moss-tts-cpp（main.cpp 的 NumPy PCG64 兼容实现）
// 编译：g++ -O2 -std=c++17 -o solution solution.cpp
#include <cstdio>
#include <cstdint>
#include <string>
#include <vector>
using namespace std;
typedef uint32_t u32;
typedef uint64_t u64;

struct U128 { u64 lo, hi; };

#if defined(__SIZEOF_INT128__)
using u128 = unsigned __int128;
static inline U128 mul128(U128 a, U128 b) {
    u128 x = ((u128)a.hi << 64 | a.lo) * ((u128)b.hi << 64 | b.lo);
    return {(u64)x, (u64)(x >> 64)};
}
#else
static inline void mul64(u64 a, u64 b, u64 &lo, u64 &hi) {
    u64 a0 = a & 0xffffffffULL, a1 = a >> 32;
    u64 b0 = b & 0xffffffffULL, b1 = b >> 32;
    u64 p00 = a0 * b0, p01 = a0 * b1, p10 = a1 * b0, p11 = a1 * b1;
    u64 mid = (p00 >> 32) + (p01 & 0xffffffffULL) + (p10 & 0xffffffffULL);
    lo = (mid << 32) | (p00 & 0xffffffffULL);
    hi = p11 + (p01 >> 32) + (p10 >> 32) + (mid >> 32);
}
static inline U128 mul128(U128 a, U128 b) {
    u64 lo, hi0, lo2, hi2, lo3, hi3;
    mul64(a.lo, b.lo, lo, hi0);
    mul64(a.hi, b.lo, lo2, hi2);
    mul64(a.lo, b.hi, lo3, hi3);
    return {lo, hi0 + lo2 + lo3};
}
#endif

static inline U128 add128(U128 a, U128 b) {
    u64 lo = a.lo + b.lo;
    return {lo, a.hi + b.hi + (lo < b.lo)};
}

static const U128 MULT = {0x4385df649fccf645ULL, 0x2360ed051fc65da4ULL};

static inline u32 hashmix(u32 v, u32 &hc) {
    v ^= hc;
    hc *= 0x931e8875u;
    v *= hc;
    v ^= v >> 16;
    return v;
}
static inline u32 mix32(u32 x, u32 y) {
    u32 r = 0xca01f9ddu * x - 0x4973f715u * y;
    r ^= r >> 16;
    return r;
}

// NumPy SeedSequence(seed)，pool_size=4，返回 PCG64 的初始 state 与 inc
static void seedseq(u64 seed, U128 &S0, U128 &INC) {
    vector<u32> ent;
    if (seed == 0) ent.push_back(0);
    else {
        u64 n = seed;
        while (n) { ent.push_back((u32)(n & 0xffffffffULL)); n >>= 32; }
    }

    u32 pool[4];
    u32 hc = 0x43b0d7e5u;                       // INIT_A
    for (int i = 0; i < 4; ++i)
        pool[i] = hashmix(i < (int)ent.size() ? ent[i] : 0u, hc);
    for (int src = 0; src < 4; ++src)
        for (int dst = 0; dst < 4; ++dst)
            if (src != dst) pool[dst] = mix32(pool[dst], hashmix(pool[src], hc));
    for (int src = 4; src < (int)ent.size(); ++src)
        for (int dst = 0; dst < 4; ++dst)
            pool[dst] = mix32(pool[dst], hashmix(ent[src], hc));

    u32 words[8];
    hc = 0x8b51f9ddu;                           // INIT_B
    for (int i = 0; i < 8; ++i) {
        u32 v = pool[i & 3];
        v ^= hc;
        hc *= 0x58f38dedu;                      // MULT_B
        v *= hc;
        v ^= v >> 16;
        words[i] = v;
    }
    u64 val[4];
    for (int j = 0; j < 4; ++j) val[j] = (u64)words[2 * j] | ((u64)words[2 * j + 1] << 32);

    U128 initstate = {val[1], val[0]};
    U128 initseq = {val[3], val[2]};
    INC = {(initseq.lo << 1) | 1u, (initseq.hi << 1) | (initseq.lo >> 63)};

    // pcg_setseq_128_srandom: state=0; step; state+=initstate; step
    U128 s = {0, 0};
    s = add128(mul128(s, MULT), INC);
    s = add128(s, initstate);
    s = add128(mul128(s, MULT), INC);
    S0 = s;
}

static inline u64 output_of(U128 s) {
    u64 x = s.hi ^ s.lo;
    unsigned r = (unsigned)(s.hi >> 58);
    if (r) x = (x >> r) | (x << (64 - r));
    return x;
}

static inline void appendUInt(string &out, u64 v) {
    char buf[24];
    int n = 0;
    if (v == 0) buf[n++] = '0';
    while (v) { buf[n++] = char('0' + int(v % 10)); v /= 10; }
    while (n) out.push_back(buf[--n]);
}

int main() {
    int Q;
    if (scanf("%d", &Q) != 1) return 0;
    string out;
    out.reserve((size_t)Q * 18);
    for (int qi = 0; qi < Q; ++qi) {
        unsigned long long seed, K;
        scanf("%llu %llu", &seed, &K);
        U128 S0, INC;
        seedseq(seed, S0, INC);

        // 该 seed 的仿射映射 A(x)=MULT*x+INC 的 2^b 次幂
        struct Aff { U128 a, c; };
        auto comp = [](const Aff &A, const Aff &B) {
            return Aff{mul128(B.a, A.a), add128(mul128(B.a, A.c), B.c)};
        };
        Aff pw[64];
        pw[0] = {MULT, INC};
        for (int b = 1; b < 64; ++b) pw[b] = comp(pw[b - 1], pw[b - 1]);

        unsigned long long steps = K + 1;   // 第 K 个输出 = 步进 K+1 次后取值
        Aff res = {{1, 0}, {0, 0}};
        for (int b = 0; b < 64; ++b)
            if ((steps >> b) & 1) res = comp(res, pw[b]);
        U128 st = add128(mul128(res.a, S0), res.c);
        appendUInt(out, output_of(st) >> 11);
        out.push_back('\n');
    }
    fputs(out.c_str(), stdout);
    return 0;
}
```
